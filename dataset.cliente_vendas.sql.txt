-- Criar tabela de clientes
CREATE TABLE Clientes (
    ClienteID INT PRIMARY KEY,
    Nome VARCHAR(50),
    Regiao VARCHAR(20)
);

-- Criar tabela de vendas
CREATE TABLE Vendas (
    VendaID INT PRIMARY KEY,
    ClienteID INT,
    Produto VARCHAR(50),
    Quantidade INT,
    Preco DECIMAL(10,2),
    Data DATE,
    FOREIGN KEY (ClienteID) REFERENCES Clientes(ClienteID)
);
